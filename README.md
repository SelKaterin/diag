# Процесс предоставления академического отпуска (AS-IS)

Диаграмма процесса (на базе Mermaid):

```mermaid
flowchart TD
    %% Пул 1: Студент
    subgraph Студент ["Пул 1: Студент"]
        direction TB
        S1([Старт])
        T1[Подача заявления]
        T2[Предоставление документов]
        I1((Ожидание результата))
        T3[Получение уведомления]
        E1([Конец])
        S1 --> T1
        T1 --> T2
        T2 --> I1
        I1 --> T3
        T3 --> E1
    end

    %% Пул 2: Единый деканат
    subgraph Деканат ["Пул 2: Единый деканат"]
        direction TB
        T4[Прием заявления и документов]
        T5[Проверка документов]
        T6[Внесение данных в 1С:Университет]
        T7[Создание приказа в 1С:Университет]
        T8[Создание приказа в 1С:Документооборот<br/>(ДУБЛИРОВАНИЕ)]
        T9[Отправка приказа на согласование]
        I2((Ожидание согласования))
        T10[Получение информации о подписании]
        T11[Ручное обновление данных в 1С]
        T12[Проведение приказа в 1С]
        T13[Уведомление студента]
        E2([Конец])

        T4 --> T5
        T5 --> T6
        T6 --> T7
        T7 --> T8
        T8 --> T9
        T9 --> I2
        I2 --> T10
        T10 --> T11
        T11 --> T12
        T12 --> T13
        T13 --> E2
    end

    %% Пул 3: Согласующие
    subgraph Согласующие ["Пул 3: Согласующие подразделения"]
        direction TB
        T14[Получение приказа на согласование]
        T15[Согласование деканом]
        T16[Согласование бухгалтерией]
        T17[Согласование отделом кадров]
        T18[Согласование проректором]
        T19[Подписание ректором]
        E3([Завершение согласования])

        T14 --> T15
        T15 --> T16
        T16 --> T17
        T17 --> T18
        T18 --> T19
        T19 --> E3
    end

    %% Связи между пулами
    T2 -->|Передача документов| T4
    T9 -->|Запуск согласования| T14
    I2 -.->|Ожидание статуса| T15
    T19 -->|Подписание завершено| T10
    T13 -->|Уведомление| T3
